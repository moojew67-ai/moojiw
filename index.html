END; $$ LANGUAGE plpgsql SECURITY DEFINER; -- STREAMING_CHUNK:Creating Shop Core Data Tables... -- ========================================== -- 3. ตารางข้อมูลร้านค้า (Shop Settings) -- ========================================== CREATE TABLE IF NOT EXISTS public.shop_settings ( id INT PRIMARY KEY DEFAULT 1 CONSTRAINT single_row_check CHECK (id = 1), name TEXT NOT NULL DEFAULT 'MooJiw Angel', cover_title TEXT DEFAULT 'Welcome Angel', cover_subtitle TEXT DEFAULT 'ศูนย์รวมไอดีเกมมือถือ สินค้าดิจิทัล รหัสขาย รหัสเช่า บริการเติมเกม 24 ชม. ปลอดภัย 100%', line_link TEXT DEFAULT 'https://line.me/R/ti/p/@moojiwangel', line_id TEXT DEFAULT '@moojiwangel', avatar_url TEXT DEFAULT 'https://images.unsplash.com/photo-1534528741775-53994a69daeb?auto=format&fit=crop&w=300&q=80', created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(), updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW() ); -- STREAMING_CHUNK:Setting Up Automatic Timestamp Triggers... -- ========================================== -- 8. ติดตั้ง Triggers อัปเดต updated_at ทุกตาราง -- ========================================== DROP TRIGGER IF EXISTS set_updated_at_shop_settings ON public.shop_settings;
